### Exercício 1: Preenchimento e exibição de uma matriz

**Objetivo:** Declare uma matriz inteira de 3 linhas e 4 colunas, preencha-a com os valores informados pelo usuário e mostre todos os seus elementos organizados em linhas e colunas.

<details>
<summary>💡 Ver Resolução / Gabarito</summary>

```cpp
#include <iostream>
using namespace std;

int main() {
    const int linhas = 3;
    const int colunas = 4;
    int matriz[linhas][colunas];

    for (int i = 0; i < linhas; ++i) {
        for (int j = 0; j < colunas; ++j) {
            cout << "Digite o elemento da linha " << i + 1
                 << " e coluna " << j + 1 << ": ";
            cin >> matriz[i][j];
        }
    }

    for (int i = 0; i < linhas; ++i) {
        for (int j = 0; j < colunas; ++j) {
            cout << matriz[i][j] << ' ';
        }
        cout << '\n';
    }
}
```

### Exercício 5: Maior elemento e sua posição

**Objetivo:** Leia uma matriz de 4 linhas e 4 colunas, encontre o maior elemento e informe todas as posições em que ele aparece.

<details>
<summary>💡 Ver Resolução / Gabarito</summary>

```cpp
#include <iostream>
using namespace std;

int main() {
    const int tamanho = 4;
    int matriz[tamanho][tamanho];

    for (int i = 0; i < tamanho; ++i) {
        for (int j = 0; j < tamanho; ++j) {
            cout << "Digite o elemento da linha " << i + 1
                 << " e coluna " << j + 1 << ": ";
            cin >> matriz[i][j];
        }
    }

    int maior = matriz[0][0];
    for (int i = 0; i < tamanho; ++i) {
        for (int j = 0; j < tamanho; ++j) {
            if (matriz[i][j] > maior) {
                maior = matriz[i][j];
            }
        }
    }

    cout << "Maior elemento: " << maior << '\n';
    for (int i = 0; i < tamanho; ++i) {
        for (int j = 0; j < tamanho; ++j) {
            if (matriz[i][j] == maior) {
                cout << "Posicao: (" << i << ", " << j << ")\n";
            }
        }
    }
}
```

### Exercício 9: Multiplicação de matrizes

**Objetivo:** Leia uma matriz `A` de 2x3 e uma matriz `B` de 3x2 e calcule a matriz resultante `C = A * B`, de dimensão 2x2.

<details>
<summary>💡 Ver Resolução / Gabarito</summary>

```cpp
#include <iostream>
using namespace std;

int main() {
    const int linhasA = 2;
    const int colunasA = 3;
    const int colunasB = 2;
    int a[linhasA][colunasA];
    int b[colunasA][colunasB];
    int resultado[linhasA][colunasB] = {};

    for (int i = 0; i < linhasA; ++i) {
        for (int j = 0; j < colunasA; ++j) {
            cout << "Digite o elemento de A na linha " << i + 1
                 << " e coluna " << j + 1 << ": ";
            cin >> a[i][j];
        }
    }

    for (int i = 0; i < colunasA; ++i) {
        for (int j = 0; j < colunasB; ++j) {
            cout << "Digite o elemento de B na linha " << i + 1
                 << " e coluna " << j + 1 << ": ";
            cin >> b[i][j];
        }
    }

    for (int i = 0; i < linhasA; ++i) {
        for (int j = 0; j < colunasB; ++j) {
            for (int k = 0; k < colunasA; ++k) {
                resultado[i][j] += a[i][k] * b[k][j];
            }
        }
    }

    for (int i = 0; i < linhasA; ++i) {
        for (int j = 0; j < colunasB; ++j) {
            cout << resultado[i][j] << ' ';
        }
        cout << '\n';
    }
}
```

</details>
