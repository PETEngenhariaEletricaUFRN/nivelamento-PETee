### Exercício 1: Sequência de múltiplos de cinco 

**Objetivo:** Preencha automaticamente, usando lógica, um vetor numérico com 10 posições com os valores `5, 10, 15, ..., 50`.

<details>
<summary>💡 Ver Resolução / Gabarito</summary>

```cpp
#include <iostream>
using namespace std;

int main() {
    const int tamanho = 10;
    int vetor[tamanho];

    for (int i = 0; i < tamanho; ++i) {
        vetor[i] = (i + 1) * 5;
    }

    for (int i = 0; i < tamanho; ++i) {
        cout << vetor[i] << ' ';
    }
    cout << '\n';
}
```

### Exercício 3: Sequência de Fibonacci em um vetor 

**Objetivo:** Preencha automaticamente, usando lógica, um vetor com 16 posições contendo os primeiros elementos da sequência de Fibonacci iniciada por `1, 1`.

<details>
<summary>💡 Ver Resolução / Gabarito</summary>

```cpp
#include <iostream>
using namespace std;

int main() {
    const int tamanho = 16;
    int vetor[tamanho] = {1, 1};

    for (int i = 2; i < tamanho; ++i) {
        vetor[i] = vetor[i - 1] + vetor[i - 2];
    }

    for (int i = 0; i < tamanho; ++i) {
        cout << vetor[i] << ' ';
    }
    cout << '\n';
}
```

### Exercício 4: Geração de números aleatórios 

**Objetivo:** Preencha automaticamente um vetor numérico com 7 números gerados aleatoriamente pelo computador e mostre os valores na tela.

<details>
<summary>💡 Ver Resolução / Gabarito</summary>

```cpp
#include <iostream>
#include <random>
using namespace std;

int main() {
    const int tamanho = 7;
    int vetor[tamanho];
    random_device semente;
    mt19937 gerador(semente());
    uniform_int_distribution<int> distribuicao(0, 99);

    for (int i = 0; i < tamanho; ++i) {
        vetor[i] = distribuicao(gerador); 
    }

    for (int i = 0; i < tamanho; ++i) {
        cout << vetor[i] << ' ';
    }
    cout << '\n';
}
```

</details>

### Exercício 5: Nomes em ordem inversa

    const int tamanho = 7;

### Exercício 8: Busca de uma chave em valores sorteados 

**Objetivo:** Preencha um vetor de 30 posições com números entre 1 e 15 sorteados pelo computador. Leia uma chave e mostre as posições em que ela foi encontrada e quantas vezes apareceu.

<details>
<summary>💡 Ver Resolução / Gabarito</summary>

```cpp
#include <iostream>
#include <random>
using namespace std;

int main() {
    const int tamanho = 30;
    int vetor[tamanho];
    random_device semente;
    mt19937 gerador(semente());
    uniform_int_distribution<int> distribuicao(1, 15);

    for (int i = 0; i < tamanho; ++i) {
        vetor[i] = distribuicao(gerador);
    }

    int chave;
    cout << "Digite a chave que deseja procurar: ";
    cin >> chave;

    int ocorrencias = 0;
    cout << "Posicoes: ";
    for (int i = 0; i < tamanho; ++i) {
        if (vetor[i] == chave) {
            cout << i << ' ';
            ++ocorrencias;
        }
    }
    cout << "\nQuantidade: " << ocorrencias << '\n';
}
```

</details>

### Exercício 9: Estatísticas de idades 

**Objetivo:** Leia a idade de 8 pessoas e mostre a média das idades, as posições com pessoas acima de 25 anos, a maior idade e as posições em que ela aparece.

<details>
<summary>💡 Ver Resolução / Gabarito</summary>

```cpp
#include <iostream>
using namespace std;

int main() {
    const int tamanho = 8;
    int idades[tamanho];
    int soma = 0;

    for (int i = 0; i < tamanho; ++i) {
        cout << "Digite a idade da pessoa " << i + 1 << ": ";
        cin >> idades[i];
        soma += idades[i];
    }

    int maior = idades[0];
    for (int idade : idades) {
        if (idade > maior) {
            maior = idade;
        }
    }

    cout << "Media: " << static_cast<double>(soma) / tamanho << '\n';
    cout << "Acima de 25 anos: ";
    for (int i = 0; i < tamanho; ++i) {
        if (idades[i] > 25) {
            cout << i << ' ';
        }
    }

    cout << "\nMaior idade: " << maior << "\nPosicoes da maior idade: ";
    for (int i = 0; i < tamanho; ++i) {
        if (idades[i] == maior) {
            cout << i << ' ';
        }
    }
    cout << '\n';
}
```

</details>

### Exercício 10: Ordenação crescente de números aleatórios 

**Objetivo:** Preencha um vetor de 20 posições com números aleatórios entre 0 e 99, mostre os valores gerados e ordene o vetor em ordem crescente implementando Bubble Sort ou Selection Sort.

<details>
<summary>💡 Ver Resolução / Gabarito</summary>

```cpp
#include <iostream>
#include <random>
using namespace std;

int main() {
    const int tamanho = 20;
    int vetor[tamanho];
    random_device semente;
    mt19937 gerador(semente());
    uniform_int_distribution<int> distribuicao(0, 99);

    for (int i = 0; i < tamanho; ++i) {
        vetor[i] = distribuicao(gerador);
    }

    cout << "Original: ";
    for (int valor : vetor) {
        cout << valor << ' ';
    }

    for (int fim = tamanho - 1; fim > 0; --fim) {
        for (int i = 0; i < fim; ++i) {
            if (vetor[i] > vetor[i + 1]) {
                int temporario = vetor[i];
                vetor[i] = vetor[i + 1];
                vetor[i + 1] = temporario;
            }
        }
    }

    cout << "\nOrdenado: ";
    for (int valor : vetor) {
        cout << valor << ' ';
    }
    cout << '\n';
}
```

</details>
