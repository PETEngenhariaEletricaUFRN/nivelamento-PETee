Exercício 6: Contagem com início, fim e incremento​

Objetivo: Crie uma função void contador(int inicio, int fim, int incremento) que mostre uma contagem a partir do início, respeitando o fim e o incremento informados.

💡 Ver Resolução / Gabarito
#include <iostream>
using namespace std;

void contador(int inicio, int fim, int incremento) {
    if (incremento <= 0) {
        cout << "O incremento deve ser positivo.\n";
        return;
    }

    for (int valor = inicio; valor <= fim; valor += incremento) {
        cout << valor << " >> ";
    }
    cout << "FIM\n";
}

int main() {
    int inicio;
    int fim;
    int incremento;

    cout << "Digite o inicio da contagem: ";
    cin >> inicio;
    cout << "Digite o fim da contagem: ";
    cin >> fim;
    cout << "Digite o incremento: ";
    cin >> incremento;
    contador(inicio, fim, incremento);
}

Exercício 7: Exibição de termos de Fibonacci​

💡 Ver Resolução / Gabarito
#include <iostream>
using namespace std;

void fibonacci(int termos) {
    int primeiro = 1;
    int segundo = 1;

    for (int i = 0; i < termos; ++i) {
        cout << primeiro << " >> ";
        int proximo = primeiro + segundo;
        primeiro = segundo;
        segundo = proximo;
    }
    cout << "FIM\n";
}

int main() {
    int termos;
    cout << "Digite a quantidade de termos de Fibonacci: ";
    cin >> termos;

    if (termos <= 0) {
        cout << "A quantidade deve ser positiva.\n";
    } else {
        fibonacci(termos);
    }
}

#include <iostream>
using namespace std;

int superSomador(int inicio, int fim) {

        fim = temporario;

    int soma = 0;

    return soma;
}

    cin >> inicio;
    cout << "Digite o fim do intervalo: ";

}

Exercício 14: Calculadora modular​

Objetivo: Crie uma calculadora separando as operações de soma, subtração, multiplicação e divisão em um cabeçalho e um arquivo de implementação. A main deve solicitar dois valores e a operação escolhida.

💡 Ver Resolução / Gabarito

operacoes.h

#ifndef OPERACOES_H
#define OPERACOES_H

double somar(double primeiro, double segundo);
double subtrair(double primeiro, double segundo);
double multiplicar(double primeiro, double segundo);
double dividir(double primeiro, double segundo);

#endif

operacoes.cpp

#include "operacoes.h"

double somar(double primeiro, double segundo) {
    return primeiro + segundo;
}

double subtrair(double primeiro, double segundo) {
    return primeiro - segundo;
}

double multiplicar(double primeiro, double segundo) {
    return primeiro * segundo;
}

double dividir(double primeiro, double segundo) {
    return primeiro / segundo;
}

main.cpp

#include <iostream>
#include "operacoes.h"
using namespace std;

int main() {
    double primeiro;
    double segundo;
    char operador;

    cout << "Digite o primeiro valor: ";
    cin >> primeiro;
    cout << "Digite o operador (+, -, * ou /): ";
    cin >> operador;
    cout << "Digite o segundo valor: ";
    cin >> segundo;

    if (operador == '/' && segundo == 0) {
        cout << "Nao e possivel dividir por zero.\n";
    } else if (operador == '+') {
        cout << "Resultado: " << somar(primeiro, segundo) << '\n';
    } else if (operador == '-') {
        cout << "Resultado: " << subtrair(primeiro, segundo) << '\n';
    } else if (operador == '*') {
        cout << "Resultado: " << multiplicar(primeiro, segundo) << '\n';
    } else if (operador == '/') {
        cout << "Resultado: " << dividir(primeiro, segundo) << '\n';
    } else {
        cout << "Operador invalido.\n";
    }
}

Exercício 15: Conversão de temperaturas​

Objetivo: Separe em arquivos uma função que converta Celsius para Fahrenheit e outra que converta Fahrenheit para Celsius. A main deve solicitar a escala e a temperatura.

💡 Ver Resolução / Gabarito

temperaturas.h

#ifndef TEMPERATURAS_H
#define TEMPERATURAS_H

double celsiusParaFahrenheit(double celsius);
double fahrenheitParaCelsius(double fahrenheit);

#endif

temperaturas.cpp

#include "temperaturas.h"

double celsiusParaFahrenheit(double celsius) {
    return celsius * 9.0 / 5.0 + 32;
}

double fahrenheitParaCelsius(double fahrenheit) {
    return (fahrenheit - 32) * 5.0 / 9.0;
}

main.cpp

#include <iostream>
#include "temperaturas.h"
using namespace std;

int main() {
    char escala;
    double temperatura;

    cout << "Digite a escala de entrada (C ou F): ";
    cin >> escala;
    cout << "Digite a temperatura: ";
    cin >> temperatura;

    if (escala == 'C' || escala == 'c') {
        cout << "Em Fahrenheit: "
             << celsiusParaFahrenheit(temperatura) << '\n';
    } else if (escala == 'F' || escala == 'f') {
        cout << "Em Celsius: "
             << fahrenheitParaCelsius(temperatura) << '\n';
    } else {
        cout << "Escala invalida.\n";
    }
}

#define GEOMETRIA_H

#endif

geometria.cpp

#include "geometria.h"

double areaCirculo(double raio) {
    const double pi = 3.141592653589793;
    return pi * raio * raio;

double volumeEsfera(double raio) {

    return 4.0 * pi * raio * raio * raio / 3.0;
}

    return primeiro + segundo;
}

double subtrairValores(double primeiro, double segundo) {
    return primeiro - segundo;
}

double multiplicarValores(double primeiro, double segundo) {
    return primeiro * segundo;
}

main.cpp

#include <iostream>
#include "menu.h"
#include "operacoes.h"
using namespace std;

int main() {
    int opcao;

    do {
        mostrarMenu();
        opcao = lerOpcao();

        if (opcao >= 1 && opcao <= 3) {
            double primeiro;
            double segundo;
            cout << "Digite o primeiro valor: ";
            cin >> primeiro;
            cout << "Digite o segundo valor: ";
            cin >> segundo;

            if (opcao == 1) {
                cout << "Resultado: "
                     << somarValores(primeiro, segundo) << '\n';
            } else if (opcao == 2) {
                cout << "Resultado: "
                     << subtrairValores(primeiro, segundo) << '\n';
            } else {
                cout << "Resultado: "
                     << multiplicarValores(primeiro, segundo) << '\n';
            }
        } else if (opcao != 0) {
            cout << "Opcao invalida.\n";
        }
    } while (opcao != 0);
}
